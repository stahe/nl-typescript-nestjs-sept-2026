// imports
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao1 {

  private axios: AxiosInstance;
  private sessionCookieName: string;
  private sessionCookie: string;

  // constructor
  constructor(axios: AxiosInstance) {
    // bibliotheek axios voor het uitvoeren van de HTTP-verzoeken
    this.axios = axios;
    // sessiecookie
    this.sessionCookieName = "PHPSESSID";
    this.sessionCookie = '';
  }

  // sessie initialiseren
  async initSession(): Promise<any> {
    // opties van het HTTP-verzoek [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      // [TypeScript-update] URL opgenomen in [options.url] (zie getRemoteData)
      url: 'main.php',
      method: "GET",
      // URL-parameters
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // uitvoeren van het HTTP-verzoek
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // opties van het HTTP-verzoek [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // body van de POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // URL-parameters
      params: {
        action: 'authenticate-user'
      }
    };
    // uitvoeren van het HTTP-verzoek
    return await this.getRemoteData(options);
  }

  // belastingberekening
  async calculateTax(married: string, children: number, salary: number): Promise<any> {
    // opties van het HTTP-verzoek
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // body van de POST
      data: qs.stringify({
        married: married,
        children: children,
        salary: salary
      }),
      // URL-parameters
      params: {
        action: 'calculate-tax'
      }
    };
    // uitvoeren van het HTTP-verzoek
    const data = await this.getRemoteData(options);
    // resultaat
    return data;
  }

  // lijst van simulaties
  async listSimulations(): Promise<any> {
    // opties van het HTTP-verzoek
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-parameters
      params: {
        action: 'list-simulations'
      },
    };
    // uitvoeren van het HTTP-verzoek
    const data = await this.getRemoteData(options);
    // resultaat
    return data;
  }

  // een simulatie verwijderen
  async deleteSimulation(index: number): Promise<any> {
    // opties van het HTTP-verzoek
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-parameters
      params: {
        action: 'delete-simulation',
        number: index
      },
    };
    // uitvoeren van het HTTP-verzoek
    const data = await this.getRemoteData(options);
    // resultaat
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // voor de sessiecookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // uitvoeren van het HTTP-verzoek
    let response: AxiosResponse;
    try {
      // asynchroon verzoek
      response = await this.axios.request(options);
    } catch (error: any) {
      // de parameter [error] is een uitzonderingsinstantie - die kan verschillende vormen aannemen
      if (error.response) {
        // het antwoord van de server staat in [error.response]
        response = error.response;
      } else {
        // we werpen de fout opnieuw
        throw error;
      }
    }
    // response is het hele HTTP-antwoord van de server (HTTP-headers + het antwoord zelf)
    // we halen de sessiecookie op als die bestaat
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie is een array
      // we zoeken de sessiecookie in deze array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // we zoeken de sessiecookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // we slaan de sessiecookie op
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // we hebben hem gevonden
          found = true;
        } else {
          // volgend element
          i++;
        }
      }
    }
    // het antwoord van de server staat in [response.data]
    return response.data;
  }
}

// export van de klasse
export default Dao1;

// asynchrone functie [main]
async function main(): Promise<void> {
  // configuratie van axios
  axios.defaults.timeout = 2000;
  axios.defaults.baseURL = 'http://localhost:3000';
  // instantiëring van de [dao]-laag
  const dao = new Dao(axios);
  // gebruik van de [dao]-laag
  try {
    // sessie initialiseren
    log("-----------init-session");
    let response = await dao.initSession();
    log(response);
    // authenticatie
    log("-----------authenticate-user");
    response = await dao.authenticateUser("admin", "admin");
    log(response);
    // belastingberekeningen
    log("-----------calculate-tax x 3");
    response = await Promise.all([
      dao.calculateTax("yes", 2, 45000),
      dao.calculateTax("no", 2, 45000),
      dao.calculateTax("no", 1, 30000)
    ]);
    log(response);
    // lijst van simulaties
    log("-----------list-of-simulations");
    response = await dao.listSimulations();
    log(response);
    // een simulatie verwijderen
    log("-----------deleting simulation no. 1");
    response = await dao.deleteSimulation(1);
    log(response);
  } catch (error: any) {
    // we loggen de fout
    console.log("error=", error.message);
  }
}

// JSON-log
function log(object: unknown): void {
  console.log(JSON.stringify(object, null, 2));
}

// uitvoering
main();

// Bedrijfsklasse
class Business {

  // [any]: de exacte structuur van de gegevens die de belastingserver verstuurt (tarieven, plafonds, ...)
  // wordt hier niet gedocumenteerd; we blijven dus bewust flexibel met dit type
  taxAdminData: any;

  // constructor
  constructor(taxAdmindata: any) {
    // this.taxAdminData: gegevens van de belastingdienst
    this.taxAdminData = taxAdmindata;
  }

  // belastingberekening
  // --------------------------------------------------------------------------
  calculateTax(married: string, children: number, salary: number): any {
    // married: ja, nee
    // children: aantal kinderen
    // salary: jaarsalaris
    // this.taxAdminData: gegevens van de belastingdienst
    //
    // belastingberekening met kinderen
    const result1 = this.calculateTax2(married, children, salary);
    const tax1 = result1["tax"];
    // belastingberekening zonder de kinderen
    let result2, tax2, halfShareCeiling;
    if (children !== 0) {
      result2 = this.calculateTax2(married, 0, salary);
      tax2 = result2["tax"];
      // toepassing van het plafond van het gezinsquotiënt
      halfShareCeiling = this.taxAdminData.halfShareQfCeiling;
      if (children < 3) {
        // HALF_SHARE_QF_CEILING euro voor de eerste 2 kinderen
        tax2 = tax2 - children * halfShareCeiling;
      } else {
        // HALF_SHARE_QF_CEILING euro voor de eerste 2 kinderen, het dubbele voor de volgende
        tax2 = tax2 - 2 * halfShareCeiling - (children - 2) * 2 * halfShareCeiling;
      }
    } else {
      // geen herberekening van de belasting
      tax2 = tax1;
      result2 = result1;
    }
    // we nemen de hoogste belasting van [tax1, tax2]
    let tax, rate, surcharge;
    if (tax1 > tax2) {
      tax = tax1;
      rate = result1["rate"];
      surcharge = result1["surcharge"];
    } else {
      surcharge = tax2 - tax1 + result2["surcharge"];
      tax = tax2;
      rate = result2["rate"];
    }
    // berekening van een eventuele korting
    const discount = this.getDiscount(married, tax);
    tax -= discount;
    // berekening van een eventuele belastingvermindering
    const reduction = this.getReduction(married, salary, children, tax);
    tax -= reduction;
    // resultaat
    return {
      "tax": Math.floor(tax), "surcharge": surcharge, "discount": discount, "reduction": reduction,
      "rate": rate
    };
  }

  // --------------------------------------------------------------------------
  calculateTax2(married: string, children: number, salary: number): any {
    // married: ja, nee
    // children: aantal kinderen
    // salary: jaarsalaris
    // this->taxAdminData: gegevens van de belastingdienst
    // 
    // aantal delen
    married = married.toLowerCase();
    let shares;
    if (married === "yes") {
      shares = children / 2 + 2;
    } else {
      shares = children / 2 + 1;
    }
    // 1 deel per kind vanaf het 3e
    if (children >= 3) {
      // een half deel extra voor elk kind vanaf het 3e
      shares += 0.5 * (children - 2);
    }
    // belastbaar inkomen
    const taxableIncome = this.getTaxableIncome(salary);
    // toeslag
    let surcharge = Math.floor(taxableIncome - 0.9 * salary);
    // vanwege afrondingsproblemen
    if (surcharge < 0) {
      surcharge = 0;
    }
    // gezinsquotiënt
    const quotient = taxableIncome / shares;
    // belastingberekening
    const limits = this.taxAdminData.limits;
    const coeffR = this.taxAdminData.coeffR;
    const coeffN = this.taxAdminData.coeffN;
    // wordt achteraan in de array limits geplaatst om de volgende lus te stoppen
    limits[limits.length - 1] = quotient;
    // zoeken naar het belastingtarief
    let i = 0;
    while (quotient > limits[i]) {
      i++;
    }
    // omdat we quotient achteraan in de array limits hebben geplaatst, kan de vorige lus
    // niet voorbij de array limits gaan
    // nu kunnen we de belasting berekenen
    const tax = Math.floor(taxableIncome * coeffR[i] - shares * coeffN[i]);
    // resultaat
    return { "tax": tax, "surcharge": surcharge, "rate": coeffR[i] };
  }

  // taxableIncome=annualSalary-allowance
  // de aftrek heeft een minimum en een maximum
  getTaxableIncome(salary: number): number {
    // aftrek van 10% van het salaris
    let allowance = 0.1 * salary;
    // deze aftrek mag niet hoger zijn dan taxAdminData.getTenPercentAllowanceMax()
    if (allowance > this.taxAdminData.tenPercentAllowanceMax) {
      allowance = this.taxAdminData.tenPercentAllowanceMax;
    }
    // de aftrek mag niet lager zijn dan taxAdminData.getTenPercentAllowanceMin()
    if (allowance < this.taxAdminData.tenPercentAllowanceMin) {
      allowance = this.taxAdminData.tenPercentAllowanceMin;
    }
    // belastbaar inkomen
    const taxableIncome = salary - allowance;
    // resultaat
    return Math.floor(taxableIncome);
  }

  // berekent een eventuele korting
  getDiscount(married: string, taxes: number): number {
    // aanvankelijk een korting van nul
    let discount = 0;
    // maximaal belastingbedrag om recht op de korting te hebben
    let taxCeilingForDiscount = married === "yes" ?
      this.taxAdminData.coupleTaxCeilingForDiscount :
      this.taxAdminData.singleTaxCeilingForDiscount;
    let discountCeiling;
    if (taxes < taxCeilingForDiscount) {
      // maximumbedrag van de korting
      discountCeiling = married === "yes" ?
        this.taxAdminData.coupleDiscountCeiling :
        this.taxAdminData.singleDiscountCeiling;
      // theoretische korting
      discount = discountCeiling - 0.75 * taxes;
      // de korting mag niet hoger zijn dan het belastingbedrag
      if (discount > taxes) {
        discount = taxes;
      }
      // geen korting <0
      if (discount < 0) {
        discount = 0;
      }
    }
    // resultaat
    return Math.ceil(discount);
  }

  // berekent een eventuele vermindering
  getReduction(married: string, salary: number, children: number, taxes: number): number {
    // het inkomensplafond om recht te hebben op de vermindering van 20%
    let incomeCeilingForReduction = married === "yes" ?
      this.taxAdminData.coupleIncomeCeilingForReduction :
      this.taxAdminData.singleIncomeCeilingForReduction;
    incomeCeilingForReduction += children * this.taxAdminData.halfShareReductionValue;
    if (children > 2) {
      incomeCeilingForReduction += (children - 2) * this.taxAdminData.halfShareReductionValue;
    }
    // belastbaar inkomen
    const taxableIncome = this.getTaxableIncome(salary);
    // vermindering
    let reduction = 0;
    if (taxableIncome < incomeCeilingForReduction) {
      // vermindering van 20%
      reduction = 0.2 * taxes;
    }
    // resultaat
    return Math.ceil(reduction);
  }
}

// export van de klasse
export default Business;

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // voor de sessiecookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // uitvoeren van het HTTP-verzoek
    let response: AxiosResponse;
    try {
      // asynchroon verzoek
      response = await this.axios.request(options);
    } catch (error: any) {
      // de parameter [error] is een uitzonderingsinstantie - die kan verschillende vormen aannemen
      if (error.response) {
        // het antwoord van de server staat in [error.response]
        response = error.response;
      } else {
        // we werpen de fout opnieuw
        throw error;
      }
    }
    // response is het hele HTTP-antwoord van de server (HTTP-headers + het antwoord zelf)
    // we halen de sessiecookie op als die bestaat
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie is een array
      // we zoeken de sessiecookie in deze array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // we zoeken de sessiecookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // we slaan de sessiecookie op
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // we hebben hem gevonden
          found = true;
        } else {
          // volgend element
          i++;
        }
      }
    }
    // het antwoord van de server staat in [response.data]
    return response.data;
  }

// imports
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao3 {

  private axios: AxiosInstance;

  // constructor
  constructor(axios: AxiosInstance) {
    this.axios = axios;
  }

  // sessie initialiseren
  async initSession(): Promise<any> {
    // opties van het HTTP-verzoek [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-parameters
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // uitvoeren van het HTTP-verzoek
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // opties van het HTTP-verzoek [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // body van de POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // URL-parameters
      params: {
        action: 'authenticate-user'
      }
    };
    // uitvoeren van het HTTP-verzoek
    return await this.getRemoteData(options);
  }

  async getAdminData(): Promise<any> {
    // opties van het HTTP-verzoek  [get /main.php?action=get-admindata]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-parameters
      params: {
        action: 'get-admindata'
      }
    };
    // uitvoeren van het HTTP-verzoek
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // uitvoeren van het HTTP-verzoek
    let response: AxiosResponse;
    try {
      // asynchroon verzoek
      response = await this.axios.request(options);
    } catch (error: any) {
      // de parameter [error] is een uitzonderingsinstantie - die kan verschillende vormen aannemen
      if (error.response) {
        // het antwoord van de server staat in [error.response]
        response = error.response;
      } else {
        // we werpen de fout opnieuw
        throw error;
      }
    }
    // response is the whole HTTP response of the server (HTTP headers + the response itself)
    // het antwoord van de server staat in [response.data]
    return response.data;
  }
}

// export van de klasse
export default Dao3;

    6.5.1. Architectuur van de oplossing

    Deze HTML-client werkt precies hetzelfde als [main3.ts], en dat is een belangrijke ontwerpkeuze: hij roept de server (PHP) slechts voor drie acties aan, steeds dezelfde: init-session, authenticate-user en get-admindata. Uit de code van de server (PHP) blijkt echter dat die ook zelf de belasting kan berekenen en de simulaties in zijn sessie kan opslaan (dit zijn de acties calculate-tax, list-simulations en delete-simulation, te vinden in de map [Controllers]), maar deze client gebruikt ze niet.

    De reden is eenvoudig: zodra de belastinggegevens (tarieven, grenzen enzovoort) één keer met get-admindata zijn opgehaald, is alles wat nodig is om de belasting te berekenen al aanwezig in de browser, dankzij de klasse [Business] . Voor elke simulatie de berekening bij de server opvragen zou een vermijdbare netwerkronde betekenen; evenzo zou het opnieuw opvragen van de lijst bij de server (list-simulations) niets toevoegen, aangezien elke berekende simulatie meteen in de browser wordt bewaard. Deze client beperkt het aantal serververzoeken dus zoveel mogelijk, precies volgens het principe dat al in [main3.ts] is toegepast:

    • de [dao]-laag ([Dao3.ts]) blijft ongewijzigd: de drie methoden die hierboven al zijn besproken (initSession, authenticateUser, getAdminData) volstaan; er komt er geen enkele bij;
    • meteen na een geslaagde authenticatie wordt getAdminData eenmalig aangeroepen en wordt het resultaat gebruikt om een object Business te maken (new Business(taxAdminData)), dat tijdens de hele werksessie bewaard blijft;
    • bij elke belastingberekening wordt vervolgens rechtstreeks business.calculateTax(married, children, salary) aangeroepen, een gewone JavaScript-functie die in de browser wordt uitgevoerd, zonder enige netwerkaanroep;
    • de berekende simulaties worden bewaard in een eenvoudige JavaScript-array die de controller in het geheugen bijhoudt: één push per berekening, één splice per verwijdering, precies zoals [main3.ts] doet met de instructies simulations.push(...) en simulations.splice(1, 1).

    Een klein verschil met [main3.ts]: de methode business.calculateTax geeft alleen de waarden terug die zij zelf heeft berekend (tax, surcharge, discount, reduction, rate), niet de parameters die de gebruiker heeft ingevoerd (married, children, salary). [main3.ts] heeft die niet nodig, omdat het gewoon console.log gebruikt; onze view [List of simulations] moet deze drie parameters echter in de tabel tonen, dus de controller voegt ze aan het resultaat toe voordat het wordt opgeslagen (dit zien we in de code van [AppHtml.ts]).

    Ten slotte doet ook de view [End of session] geen aanroep meer naar de server: die gooit simpelweg het object [Business] en de lijst met simulaties weg en toont daarna het authenticatiescherm opnieuw. Houd er rekening mee dat de antwoorden van de drie acties die nog worden gebruikt altijd hetzelfde formaat hebben, dat we al in [main3.ts] hebben gezien: een JSON-object { action, status, response }, waarvan response ofwel het gevraagde resultaat bevat (bij succes), ofwel een foutmelding: een tekenreeks of een array van tekenreeksen, afhankelijk van de betreffende PHP-controller.

    6.5.1.4. De laag [views]: het bestand [Views.ts]

    Dit is het langste bestand van deze client, maar ook het gemakkelijkst te begrijpen zodra u het principe doorhebt: de klasse [Views] doet geen enkele netwerkaanroep; ze weet niet eens dat de PHP-server bestaat. Haar enige taak is om met de DOM-functies (document.createElement, appendChild, classList...) de HTML-elementen van de drie schermen van de applicatie op te bouwen binnen de tag <div id="app"> van de template.

    Wanneer de gebruiker op een knop of een koppeling klikt („Submit”, „Delete”, „End of session” enzovoort), weet de klasse [Views] ook niet wat ze met die klik moet doen: ze roept alleen een callbackfunctie aan die de controller aanlevert. Het is de controller, [AppHtml.ts], die hierna wordt besproken, die vervolgens beslist wat er moet gebeuren (de server aanroepen, van view wisselen enzovoort). Dankzij deze ontkoppeling kan [Views.ts] onafhankelijk blijven van de [dao]-laag.

      div.className = 'help';
      div.textContent = help;
      field.appendChild(div);

    • regels 333 tot en met 339: de koppeling „Delete” van elke rij; de klikhandler (regel 337) roept callbacks.onDelete(number) aan met het nummer van de betreffende rij.

    6.5.1.5. De controller: het bestand [AppHtml.ts]

    Dit is de orkestrator van de applicatie: hij neemt de rol over van de functie main() in **[main3.ts], maar in de vorm van een klasse met één methode per view. Hij coördineert nu twee lagen, [dao] (de server) en [business] (de berekening), voordat hij de [views]**-laag opdracht geeft om ofwel een resultaat ofwel een foutmelding te tonen:

    
// imports
import axios from 'axios';
import "core-js/stable";
import "regenerator-runtime/runtime";

// imports
import Dao from './Dao3';
import Business from './Business';
import Views, { Simulation } from './Views';

// -----------------------------------------------------------------------
// controller van de HTML-applicatie: stuurt de aanroepen naar de server
// (laag [dao]), de berekeningen (laag [business]) en de weergave van de views
// (laag [views]) aan.
//
// Net als [main3.ts] roept deze client de PHP-server maar voor drie
// acties aan: [init-session], [authenticate-user] en [get-admindata].
// Zodra de belastinggegevens (tarieven, grenzen...) met
// [get-admindata] zijn opgehaald, berekent de laag [business] (klasse [Business], hetzelfde
// bestand [Business.ts] als bij [main3.ts]) de belasting
// in de browser, zonder nieuwe netwerkaanroep. De lijst met simulaties wordt ook
// alleen in de browser bijgehouden (array [this.simulations]),
// precies zoals [main3.ts] doet met zijn lokale array [simulations]
// (zie `simulations.push(...)` en `simulations.splice(...)`).
// -----------------------------------------------------------------------
class AppHtml {

  private dao: Dao;
  private views: Views;
  // [business]-laag: bestaat pas zodra de belastinggegevens zijn opgehaald
  // (na een geslaagde authenticatie) - zie [initBusiness]
  private business?: Business;
  // lijst met de al berekende simulaties, in de browser bewaard
  private simulations: Simulation[] = [];

  constructor() {
    // configuratie van axios - identiek aan [main3.ts]
    axios.defaults.timeout = 5000;
    axios.defaults.baseURL = 'http://localhost/';
    axios.defaults.withCredentials = true;
    // [dao]- en [views]-lagen
    this.dao = new Dao(axios);
    this.views = new Views(document.getElementById('app')!);
  }

  // startpunt van de applicatie
  async start(): Promise<void> {
    try {
      // sessie initialiseren - verplicht vóór elke andere actie (zie [main.php])
      const response = await this.dao.initSession();
      if (response.status !== 700) {
        this.views.authenticationView(this.formatError(response.response), {
          onSubmit: (user, password) => this.authenticate(user, password)
        });
        return;
      }
    } catch (error: any) {
      this.views.authenticationView(
        "Unable to contact the server (" + error.message + ")",
        { onSubmit: (user, password) => this.authenticate(user, password) }
      );
      return;
    }
    // sessie geïnitialiseerd: we tonen de authenticatie
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // view [authenticatie]
  // --------------------------------------------------------------------
  private showAuthentication(error?: string): void {
    this.views.authenticationView(error, {
      onSubmit: (user, password) => this.authenticate(user, password)
    });
  }

  private async authenticate(user: string, password: string): Promise<void> {
    try {
      const response = await this.dao.authenticateUser(user, password);
      if (response.status === 200) {
        // geslaagde authenticatie: zoals in [main3.ts] halen we nu,
        // eens en voor altijd, de belastinggegevens op
        // die de [business]-laag nodig heeft
        await this.initBusiness();
      } else {
        // mislukking (verkeerde login/wachtwoord, of ontbrekende parameters)
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // haalt de gegevens van de belastingdienst op en instantieert de
  // [business]-laag met die gegevens - één enkele serveraanroep voor de hele sessie,
  // precies zoals [main3.ts]
  private async initBusiness(): Promise<void> {
    try {
      const response = await this.dao.getAdminData();
      if (response.status === 1000) {
        this.business = new Business(response.response);
        // nieuwe werksessie: we beginnen opnieuw met een lege lijst
        this.simulations = [];
        this.showCalculation();
      } else {
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // --------------------------------------------------------------------
  // view [belastingberekening]
  // --------------------------------------------------------------------
  private showCalculation(result?: Simulation, error?: string): void {
    this.views.calculationView(result, error, {
      onSubmit: (married, children, salary) => this.calculate(married, children, salary),
      onList: () => this.showList(),
      onEndSession: () => this.endSession()
    });
  }

  // belastingberekening volledig in de browser uitgevoerd ([business]-laag):
  // geen aanroep naar de server, anders dan bij de eerste versie van deze client
  private calculate(married: string, children: number, salary: number): void {
    // this.business is noodzakelijkerwijs geïnitialiseerd: we komen pas in de
    // view [berekening] na [initBusiness]
    const result = this.business!.calculateTax(married, children, salary);
    // het door [business.calculateTax] teruggegeven resultaat bevat alleen de
    // berekende waarden (tax, surcharge, discount, reduction, rate): we voegen daar de
    // door de gebruiker ingevoerde parameters aan toe, zodat ze opnieuw
    // kunnen worden weergegeven in de lijst met simulaties
    const simulation: Simulation = { married: married, children: children, salary: salary, ...result };
    // opslag in de browser (zoals `simulations.push(...)` in [main3.ts])
    this.simulations.push(simulation);
    this.showCalculation(simulation);
  }

  // --------------------------------------------------------------------
  // view [lijst met simulaties]
  // --------------------------------------------------------------------
  // hier geen serveraanroepen meer: de lijst staat al volledig in
  // [this.simulations]
  private showList(error?: string): void {
    this.views.listView(this.simulations, error, {
      onDelete: (number) => this.deleteSimulation(number),
      onCalculation: () => this.showCalculation(),
      onEndSession: () => this.endSession()
    });
  }

  private deleteSimulation(number: number): void {
    // lokale verwijdering uit de array [this.simulations], zoals
    // `simulations.splice(1, 1)` in [main3.ts] - nog steeds geen serveraanroep
    this.simulations.splice(number, 1);
    this.showList();
  }

  // --------------------------------------------------------------------
  // einde van de sessie
  // --------------------------------------------------------------------
  private endSession(): void {
    // einde van de sessie alleen aan de clientkant: we vergeten de [business]-laag
    // (en dus de belastinggegevens) en de al berekende simulaties, daarna gaan we
    // terug naar het authenticatiescherm. Deze client hoeft hiervoor de server niet
    // aan te roepen.
    this.business = undefined;
    this.simulations = [];
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // formatteert het veld [response] van een foutantwoord van de server,
  // dat een tekenreeks of een array van tekenreeksen is (zie de PHP-controllers)
  // --------------------------------------------------------------------
  private formatError(response: unknown): string {
    if (Array.isArray(response)) {
      return response.join(' ; ');
    }
    return String(response);
  }
}

// uitvoering
window.addEventListener('DOMContentLoaded', () => {
  new AppHtml().start();
});

    6.5.1.6. Compilatie en uitvoering

    Compilatie en uitvoering gebeuren precies zoals in de vorige paragraaf ([npm run build] of [npm start]); alleen het ingangspunt in [webpack.config.js] is gewijzigd. Hier volgen de drie schermen die dit oplevert, om te vergelijken met de schermafbeeldingen uit het document [web_application_views.odt] dat aan het begin is meegeleverd:

    Authenticatiescherm: de gebruikersnaam is voor het gemak vooraf ingevuld met „admin”; als de authenticatie mislukt, verschijnt hier boven het formulier een foutmelding.

    Scherm voor de belastingberekening, na het verzenden van het eerste formulier: het groene vak toont het resultaat dat in de browser is berekend door de [business]-laag (tax, discount, reduction, surcharge, rate, hier omgezet naar een percentage).

    Scherm „List of your simulations”, bereikt door te klikken op de koppeling „List of simulations”: elke rij komt overeen met een simulatie die in de tabel [this.simulations] van de browser is opgeslagen, met bijbehorende koppeling „Delete”.

    De vijf views zijn de volgende: authenticatie, berekening met resultaten, lijst van simulaties, verwijderen van een simulatie (waarna de bijgewerkte lijst opnieuw wordt getoond) en einde van de sessie (waarna het authenticatiescherm weer verschijnt); dit alles zonder ooit verzoeken aan de PHP-server te doen buiten de drie acties init-session, authenticate-user en get-admindata.
