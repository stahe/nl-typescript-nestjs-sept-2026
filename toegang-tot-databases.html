// Eerste stap: verbinding maken met de MySQL-database [dbpersons] en daarna
// de verbinding verbreken. Port van het Python-script [mysql_01] van de cursus Python/Flask.
import mysql from 'mysql2/promise';

// verbinding met de MySQL-database [dbpersons]
// de identiteit van de gebruiker is (admpersons, nobody)
const USER = 'admpersons';
const PWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// daar gaan we
let connection: mysql.Connection | undefined;
try {
  console.log('Connecting to the MySQL DBMS...');
  // verbinding
  connection = await mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE });
  // vervolg
  console.log(
    `Successful MySQL connection to database=${DATABASE}, host=${HOST} as user=${USER}, passwd=${PWD}`,
  );
} catch (error) {
  // de fout tonen
  console.log(`The following error occurred: ${error}`);
} finally {
  // de verbinding sluiten als die was geopend
  if (connection) {
    await connection.end();
  }
}

    Laten we deze code toelichten:

    • regel 16: [mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE })] opent één enkele, asynchrone verbinding; createConnection geeft een Promise<Connection> terug, vandaar het sleutelwoord await; dit is het TypeScript-equivalent van mysql.connector.connect(...) aan de Python-kant;
    • regel 29: [connection.end()] sluit de verbinding netjes in het blok finally, ongeacht of er een fout is opgetreden; de verbinding is buiten het blok try gedeclareerd (type mysql.Connection | undefined) om op dit niveau bereikbaar te blijven.

    // databases/mysql/mysql-02.ts
// In dit nieuwe script is de databaseverbinding geïsoleerd in een functie.
// Port van het Python-script [mysql_02].
import mysql from 'mysql2/promise';

// maakt verbinding en verbreekt die daarna (login, pwd) met de database [database] op de server [host]
// werpt een uitzondering bij een probleem
async function connect(host: string, database: string, login: string, pwd: string): Promise<void> {
  let cnx: mysql.Connection | undefined;
  try {
    // verbinding
    cnx = await mysql.createConnection({ host, user: login, password: pwd, database });
    console.log(
      `Successful connection to database=${database}, host=${host} as user=${login}, passwd=${pwd}`,
    );
  } finally {
    // de verbinding sluiten als die was geopend
    if (cnx) {
      await cnx.end();
      console.log('Successful disconnection\n');
    }
  }
}

// main
// inloggegevens voor de verbinding
const USER = 'admpersons';
const PASSWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// verbinding van een bestaande gebruiker
try {
  await connect(HOST, DATABASE, USER, PASSWD);
} catch (error) {
  // de fout tonen
  console.log(String(error));
}

// verbinding van een niet-bestaande gebruiker
try {
  await connect(HOST, DATABASE, 'xx', 'xx');
} catch (error) {
  // de fout tonen
  console.log(String(error));
}

    Laten we deze code toelichten:

    • regel 9: [async function connect(host, database, login, pwd): Promise<void>] is een async-functie die niets teruggeeft maar wel een uitzondering kan opwerpen; ze vangt verbindingsfouten niet zelf af, maar laat ze doorstromen naar de aanroeper;
    • regel 40: [await connect(HOST, DATABASE, 'xx', 'xx')] is de tweede aanroep, met een gebruiker „xx” die aan de MySQL-kant niet bestaat; dit werpt een uitzondering op die het omsluitende catch-blok weergeeft.

    Versiespecifiek probleem: op een recente MySQL-server (8.4 en hoger) kan deze tweede aanroep een andere melding opleveren: „Error: Plugin 'mysql_native_password' is not loaded” in plaats van „Access denied for user 'xx'@'localhost'.” De oorzaak ligt niet bij het script: om niet te verklappen of een account bestaat (bescherming tegen enumeratie) simuleert MySQL een authenticatie-uitwisseling bij het inloggen met een onbekende gebruiker, en die simulatie is vast gekoppeld aan de verouderde plug-in mysql_native_password, die sinds MySQL 8.4 standaard is uitgeschakeld.     // databases/mysql/mysql-module.ts
// Herbruikbare functies om een lijst (of een bestand) met SQL-opdrachten uit te voeren
// op een geopende MySQL-verbinding. Port van de Python-module [mysql_module].
import type { Connection, FieldPacket, ResultSetHeader, RowDataPacket } from 'mysql2/promise';
import { readFile } from 'node:fs/promises';

// toont het resultaat van een SQL-opdracht (resultaat van een query() van mysql2)
// - als er een array [fields] is, was het een SELECT: we tonen de kolommen
//   en daarna de rijen;
// - anders was het een wijzigingsopdracht (INSERT/UPDATE/DELETE/DDL): we
//   tonen het aantal gewijzigde rijen (equivalent van cursor.rowcount).
function displayInfo(rows: RowDataPacket[] | ResultSetHeader, fields: FieldPacket[] | undefined): void {
  if (fields && fields.length > 0) {
    // het was een SELECT — fields beschrijft de gevraagde kolommen
    const titles = fields.map((f) => f.name);
    console.log(titles.join(', '));
    console.log('*'.repeat(titles.join(', ').length));
    for (const row of rows as RowDataPacket[]) {
      console.log(titles.map((t) => row[t]));
    }
    console.log('*'.repeat(titles.join(', ').length));
  } else {
    // geen beschreven kolommen — wijzigingsopdracht
    const header = rows as ResultSetHeader;
    console.log(`number of modified rows: ${header.affectedRows}`);
  }
}

// gebruikt de geopende verbinding [connection]
// voert op deze verbinding de SQL-opdrachten uit die in de lijst [sqlCommands] staan
// (één opdracht per element; lege regels of regels die met # beginnen worden genegeerd)
// - als verbose=true, wordt na elke uitvoering van een SQL-opdracht een melding getoond
//   die het slagen of mislukken aangeeft;
// - als stop=true, stopt de functie bij de 1e fout, anders
//   voert ze alle opdrachten uit;
// - als withTransaction=true, maakt elke fout alle eerder uitgevoerde SQL-opdrachten ongedaan
//   (rollback); anders wordt elke opdracht
//   afzonderlijk gecommit (standaardgedrag van mysql2);
// de functie geeft een array [error1, error2, ...] terug
export async function executeListOfCommands(
  connection: Connection,
  sqlCommands: string[],
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  const errors: string[] = [];
  if (withTransaction) {
    await connection.beginTransaction();
  }
  try {
    for (const rawCommand of sqlCommands) {
      // verwijder de voorafgaande en volgende witruimte van de huidige opdracht
      const command = rawCommand.trim();
      // lege opdracht of commentaar? ga naar de volgende
      if (command === '' || command[0] === '#') {
        continue;
      }
      // uitvoering van de huidige opdracht
      try {
        const [rows, fields] = await connection.query(command);
        // geen fout
        if (verbose) {
          console.log(`[${command}]: Successful execution`);
        }
        displayInfo(rows as RowDataPacket[], fields);
      } catch (error) {
        const msg = `${command}: Error (${error})`;
        errors.push(msg);
        if (verbose) {
          console.log(msg);
        }
        // stoppen we?
        if (withTransaction || stop) {
          return errors;
        }
      }
    }
    return errors;
  } finally {
    // de transactie committen / terugdraaien als er een is
    if (withTransaction) {
      if (errors.length > 0) {
        await connection.rollback();
      } else {
        await connection.commit();
      }
    }
  }
}

// gebruikt de geopende verbinding [connection]
// voert op deze verbinding de SQL-opdrachten uit die in het
// tekstbestand [sqlFilename] staan (één opdracht per regel)
export async function executeFileOfCommands(
  connection: Connection,
  sqlFilename: string,
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  try {
    const content = await readFile(sqlFilename, 'utf-8');
    return await executeListOfCommands(connection, content.split('\n'), verbose, stop, withTransaction);
  } catch (error) {
    return [`The file ${sqlFilename} could not be processed: ${error}`];
  }
}

    Laten we deze code toelichten:

    • regel 45: [if (withTransaction) { await connection.beginTransaction(); }] : mysql2 staat standaard in de modus AUTOCOMMIT (elke query wordt afzonderlijk bevestigd); beginTransaction() zet de verbinding expliciet over op transactiemodus, net zoals connexion.start_transaction() dat aan de Python-kant zou doen;
    • regel 76: [if (withTransaction || stop) { return errors; }] : in transactiemodus beëindigt de eerste fout de lus onmiddellijk; doorgaan zou zinloos zijn, aangezien de hele transactie toch wordt teruggedraaid;
    • regel 82: [if (errors.length > 0) { await connection.rollback(); } else { await connection.commit(); }] : globale commit of rollback, afhankelijk van of er een fout is opgetreden; zie echter de valkuil die wordt beschreven bij het script [mysql-04] hieronder.

// Geparametriseerde queries: eerst een enkele voorbereide invoeging, daarna bulkinvoeging.
// Port van het Python-script [mysql_05]. Hier maken we de tabel [persons] opnieuw aan met
// het schema met een kolom [id] (zoals in script mysql-03), zodat dit script
// onafhankelijk is van de toestand die mysql-04 achterlaat.
import mysql from 'mysql2/promise';

// de identiteit van de gebruiker
const ID = 'admpersons';
const PWD = 'nobody';
// de hostmachine van het DBMS
const HOST = 'localhost';
// identiteit van de database
const DATABASE = 'dbpersons';

type Person = [id: number, lastName: string, firstName: string, age: number];

// lijst van personen (id, lastName, firstName, age)
const persons: Person[] = [];
for (let i = 0; i < 5; i++) {
  persons.push([i, `l0${i}`, `f0${i}`, i + 10]);
}
persons.push([40, "d'Aboot", "Y'éna", 18]);

// nog een lijst van personen
const otherPersons: Person[] = [];
for (let i = 0; i < 5; i++) {
  otherPersons.push([i + 100, `l1${i}`, `f1${i}`, i + 20]);
}
otherPersons.push([200, "d'Aboot", "F'ilhem", 34]);

// toegang tot het DBMS
let connection: mysql.Connection | undefined;
try {
  // verbinding
  connection = await mysql.createConnection({ host: HOST, user: ID, password: PWD, database: DATABASE });

  // (her)aanmaak van de tabel met het schema met de kolom [id] (zie mysql-03.ts)
  await connection.query('drop table if exists persons');
  await connection.query(
    'create table persons (id int PRIMARY KEY, first_name varchar(30) NOT NULL, last_name varchar(30) NOT NULL, ' +
      'age integer NOT NULL, unique(last_name,first_name))',
  );

  // verwijdering van de bestaande records
  await connection.query('delete from persons');
  // invoegingen persoon voor persoon met een voorbereide instructie
  for (const person of persons) {
    const [id, lastName, firstName, age] = person;
    await connection.execute('insert into persons(id,last_name,first_name,age) values(?,?,?,?)', [id, lastName, firstName, age]);
  }
  // bulkinvoeging van een lijst van personen — equivalent van executemany:
  // mysql2 biedt connection.query('insert ... values ?', [[...],[...],...])
  // waarmee één query met meerdere rijen wordt opgebouwd
  const values = otherPersons.map(([id, lastName, firstName, age]) => [id, lastName, firstName, age]);
  await connection.query('insert into persons(id,last_name,first_name,age) values ?', [values]);

  // committen van de transactie — hier is elke query al automatisch gecommit
  // (mysql2 staat standaard in de modus AUTOCOMMIT, zoals de standaardmodus die in
  // script mysql-03 is genoemd; er valt dus niets expliciet te committen)
  console.log(`${persons.length + otherPersons.length} persons inserted.`);
} catch (error) {
  console.log(`The following error occurred: ${error}`);
} finally {
  if (connection) {
    await connection.end();
  }
}

    7.2. Het ORM TypeORM gebruiken

    We behandelen nu ORM, op basis van het hoofdstuk „Using the SQLAlchemy ORM” (hoofdstuk 19) uit de cursus Python/Flask. Het principe van ORM (object-relational mapping) is dat u objecten van de taal manipuleert in plaats van SQL te schrijven: een klasse komt overeen met een tabel, een instantie met een rij, een attribuut met een kolom en een relatie tussen klassen (associatie, compositie) met een refererende sleutel.

    Twee ORM-implementaties worden na elkaar gepresenteerd, met hetzelfde voorbeeld, zodat ze kunnen worden vergeleken: TypeORM in dit hoofdstuk, gevolgd door Prisma in het volgende. Van de twee lijkt TypeORM qua stijl het meest op SQLAlchemy (gedecoreerde klassen, Repository die met methoden wordt bevraagd); Prisma daarentegen genereert een volledig getypeerde client op basis van een apart, declaratief schemabestand.

// Berekent voor een gegeven leerling het gemiddelde, gewogen met de coëfficiënten van
// de vakken, en het laagste en hoogste cijfer. Neemt het Python-script
// [main_stats_for_student] (hoofdstuk 19, scripts 05) over.
//
// Valkuil: de `decimal` kolommen (hier `grade.value` en
// `subject.coefficient`) hebben in de entiteit aan de TypeScript-kant het type `number`,
// maar de mysql2-driver levert ze in werkelijkheid als
// *tekenreeksen* (om geen precisie te verliezen bij
// omzetting naar een `number` met drijvende komma). `typeof grade.value` is tijdens
// runtime `'string'`, ondanks het gedeclareerde type. Een berekening zoals
// `sum += grade.value * subject.coefficient` zou dan een
// aaneenschakeling of een riskante impliciete omzetting uitvoeren: u moet
// vóór elke berekening expliciet omzetten met `Number(...)`.
import { AppDataSource } from './data-source-school.js';
import { Student } from './entities/student.entity.js';

// berekent de statistieken (gewogen gemiddelde, minimum, maximum) van een leerling
async function studentStats(studentId: number): Promise<void> {
  const studentRepo = AppDataSource.getRepository(Student);
  // we laden de leerling met zijn cijfers en, per cijfer, het bijbehorende
  // vak (voor de coëfficiënt) — één query met twee
  // joins, dankzij de geneste optie `relations`
  const student = await studentRepo.findOneOrFail({
    where: { id: studentId },
    relations: { grades: { subject: true } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // expliciete omzetting: `grade.value` en `grade.subject.coefficient`
    // zijn tijdens runtime tekenreeksen, ondanks hun gedeclareerde type `number`
    const value = Number(grade.value);
    const coefficient = Number(grade.subject.coefficient);
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;

  }

  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

    7.2.7. Conclusie

    TypeORM maakt het mogelijk lees- en schrijfbewerkingen zonder SQL te schrijven, met gedecoreerde entiteiten die qua opzet op SQLAlchemy lijken, ten koste van een belangrijk ontwerpverschil waarmee u rekening moet houden (geen impliciete lazy loading van relaties: alle relaties die niet expliciet worden opgevraagd, blijven weg) en een tooling-valkuil die specifiek is voor uitvoering met TSX (kolomtypen moeten expliciet worden opgegeven). Het volgende hoofdstuk gebruikt exact hetzelfde voorbeeld met Prisma om de twee benaderingen te vergelijken.

    7.3. Het ORM Prisma gebruiken

    Tweede ORM, met hetzelfde voorbeeld als in het vorige hoofdstuk (de databases dbpersons en dbschool), om de twee benaderingen te vergelijken. Prisma verschilt qua ontwerp van TypeORM: in plaats van klassen te annoteren, wordt het databaseschema beschreven in een apart bestand met het eigen declaratieve formaat van Prisma, waaruit de opdracht prisma generate een volledig getypeerde client genereert.

    Belangrijk: op het moment van schrijven van dit hoofdstuk is Prisma ORM versie 7, die ingrijpende interne wijzigingen heeft ondergaan ten opzichte van eerdere versies (5 en 6): de queryengine, voorheen een apart gedownload binair bestand dat in Rust was gecompileerd, is nu een WebAssembly-module die rechtstreeks in het npm-pakket is ingebed, gecombineerd met een driveradapter (hier @prisma/adapter-mariadb) die queries doorgeeft aan een traditionele JavaScript-driver. Het algemene principe blijft echter hetzelfde als bij eerdere versies van Prisma, en het volgende geldt dus ook voor die versies.

    Opmerking over deze omgeving: de opdracht npx prisma generate , die de TypeScript-client genereert uit het bestand schema.prisma, vereist een Prisma-specifieke netwerkbron (binaries.prisma.sh) om een intern onderdeel van de CLI op te halen; het netwerkbeleid van de uitvoeringsomgeving waarin deze cursus is voorbereid, stond geen toegang tot die bron toe. De scripts in dit hoofdstuk zijn daarom met dezelfde zorg geschreven en nagelezen als de vorige, maar konden in deze omgeving niet worden uitgevoerd of geverifieerd, anders dan die in de twee vorige hoofdstukken, die allemaal zijn uitgevoerd en waarvan de resultaten zijn geverifieerd. Op een machine met normale internettoegang (het gebruikelijke geval) npx prisma generate werkt zonder bijzondere problemen, en de scripts zouden de hieronder getoonde „verwachte” uitvoer moeten opleveren, berekend uit dezelfde gegevens die in het vorige hoofdstuk al zijn geverifieerd.

  // equivalent van `relations: { grades: { subject: true } }` van TypeORM
  console.log('grades of student id=11 -----------');
  const studentWithGrades = await prisma.student.findUniqueOrThrow({
    where: { id: 11 },

  for (const grade of studentWithGrades.grades) {

  for (const className of ['class1', 'class2']) {

  // de cijfers voor een vak, op vaknaam
  for (const subjectName of ['subject1', 'subject2']) {
    console.log(`subject with name='${subjectName}' -----------`);
    const subject = await prisma.subject.findUniqueOrThrow({ where: { name: subjectName }, include: { grades: true } });
    console.log(`subject={"name":"${subject.name}","coefficient":${subject.coefficient}}`);
    console.log('Grades in the subject: ');
    for (const grade of subject.grades) {
      console.log(`{"id":${grade.id},"value":${grade.value}}`);
    }
  }
} finally {
  await prisma.$disconnect();
}

// Berekent voor een gegeven leerling het gemiddelde, gewogen met de coëfficiënten van
// de vakken, en het laagste en hoogste cijfer. Gebruikt hetzelfde voorbeeld als
// databases/typeorm/typeorm-04-stats.ts.
//
// Valkuil die verschilt van die van TypeORM: Prisma levert de
// `Decimal` kolommen (hier `grade.value` en `subject.coefficient`) niet als
// tekenreeksen, maar als instanties van de klasse `Prisma.Decimal`
// (geleverd door de bibliotheek decimal.js), om geen enkele
// decimale precisie te verliezen. `typeof grade.value` is `'object'`, niet `'number'`
// en ook niet `'string'`. Een berekening zoals `grade.value * subject.coefficient`
// zou bijna toevallig werken (`Decimal` herdefinieert `valueOf`), maar
// de goede praktijk — en de praktijk die hieronder wordt gevolgd — is om
// vóór elke berekening met drijvende komma expliciet `.toNumber()` aan te roepen.
import { PrismaMariaDb } from '@prisma/adapter-mariadb';
import { PrismaClient } from '../../generated/prisma-school/client.js';

const adapter = new PrismaMariaDb({
  host: 'localhost',
  user: 'admschool',
  password: 'pwdschool',
  database: 'dbschool',
});
const prisma = new PrismaClient({ adapter });

// berekent de statistieken (gewogen gemiddelde, minimum, maximum) van een leerling
async function studentStats(studentId: number): Promise<void> {
  // één enkele query, met twee niveaus van join (cijfers, daarna het vak van
  // elk cijfer) om de coëfficiënt van elk vak te hebben
  const student = await prisma.student.findUniqueOrThrow({
    where: { id: studentId },
    include: { grades: { include: { subject: true } } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // expliciete omzetting: `grade.value` en `grade.subject.coefficient`
    // zijn waarden van `Prisma.Decimal`, geen waarden van `number`
    const value = grade.value.toNumber();
    const coefficient = grade.subject.coefficient.toNumber();
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;
    min = Math.min(min, value);
    max = Math.max(max, value);
    console.log(`  grade=${value.toFixed(2)} subject="${grade.subject.name}" (coefficient ${coefficient.toFixed(2)})`);
  }

  const average = weightedSum / coefficientsSum;
  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

try {
  // leerling id=11: cijfers subject1=10 (coëff. 1.0), subject2=6 (coëff. 2.0)
  // gewogen gemiddelde = (10*1.0 + 6*2.0) / (1.0+2.0) = 22/3 = 7.33
  await studentStats(11);
  console.log();
  await studentStats(21);
  console.log();
  await studentStats(32);
  console.log();
  await studentStats(42);
} finally {
  await prisma.$disconnect();
}
