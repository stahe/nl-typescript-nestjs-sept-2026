    5. De HTTP-functies van TypeScript

    Hier presenteren we twee manieren om HTTP-verzoeken te doen in TypeScript: de functie [fetch], die sinds versie 18 native in Node aanwezig is, en de bibliotheek van derden [axios]. De scripts van dit hoofdstuk staan in de map [http] van het project.

    5.1. HTTP-hulpmiddelen kiezen

    In 2019 was fetch nog niet geïmplementeerd door node.js en het oorspronkelijke document raadde het pakket node-fetch aan. Sinds Node 18 (2022) is [fetch] een native globale functie, gestandaardiseerd door de WHATWG Fetch-standaard, die niet meer hoeft te worden geïmporteerd.

    [axios] blijft echter een interessante bibliotheek: ze is compatibel met zowel node.js als browsers en biedt een iets rijkere API (interceptors, automatische JSON-serialisatie, gestandaardiseerde foutafhandeling). We presenteren beide aan de hand van hetzelfde voorbeeld om te laten zien dat de programmeeraanpak vergelijkbaar is.

    5.2. De server voor de belastingberekening

    Dit hoofdstuk verwijst, net als de volgende, naar de eerder geïnstalleerde NestJS-server. De testarchitectuur is als volgt:

     

    In [A-B] verstuurt JavaScript-code die door [node.js] wordt uitgevoerd verzoeken naar de webapplicatie [C] via [1]. De webapplicatie stuurt JSON-antwoorden terug (4). De JavaScript-code aan de clientzijde kan in lagen worden georganiseerd [UI, business, dao] (UI: gebruikersinterface). Dat doen we later. Om de volgende scripts uit te voeren, moet uw NestJS-server draaien.

// [update 2026] in 2019 moest u het pakket [node-fetch] installeren om de functie
// [fetch] aan de node-kant te hebben. Sinds Node 18 (2022) is [fetch] een native globale
// functie, gestandaardiseerd door de "WHATWG Fetch standard": importeren is niet meer nodig.
// imports
import qs from 'qs';
import { sprintf } from 'sprintf-js';
import moment from 'moment';


// basis-URL van de server voor de belastingberekening
const baseUrl: string = 'http://localhost:3000/main.php?';
// sessie initialiseren
async function initSession(): Promise<any> {
  // [update 2026] de native [fetch] kent de optie [timeout] niet.
  // om de wachttijd te beperken geven we hem een [signal] die door
  // [AbortSignal.timeout(ms)] wordt geproduceerd (beschikbaar sinds Node 17.3): na die tijd
  // wordt het verzoek automatisch geannuleerd en werpt [fetch] een fout.
  const options: RequestInit = {
    method: "GET",
    signal: AbortSignal.timeout(2000)
  };
  // uitvoering van het HTTP-verzoek [get /main.php?action=init-session&type=json]
  let fetchStart: moment.Moment | undefined;
  try {
    // asynchroon verzoek - [fetch] geeft een promise terug
    fetchStart = moment(Date.now());
    const response = await fetch(baseUrl + qs.stringify({
      action: 'init-session',
      type: 'json'
    }), options);
    // [response] is het volledige HTTP-antwoord van de server (HTTP-headers + het antwoord zelf)
    // we tonen dit antwoord om de structuur ervan te zien
    console.log(sprintf("fetch response formatted as json,=%j, %s", response, time(fetchStart)));
    console.log("fetch response in javascript=", response);
    // we hebben toegang tot de HTTP-headers
    console.log("response headers=", response.headers);
    // als het antwoord van het type application / json is, wordt het json-antwoord van de server verkregen met de asynchrone functie [response.json()]
    // in dat geval krijgt de aanroepende code een object [Promise]
    // dankzij [await] krijgt u het [json]-antwoord van de server in plaats van zijn promise
    const jsonStart = moment(Date.now());
    const object = await response.json();
    console.log(sprintf("json response=%j, type=%s, %s", object, typeof (object), time(jsonStart)));
    return object;
    // als het antwoord van het type text / plain is, wordt het tekstantwoord van de server verkregen met [response.text()]
    // in dat geval krijgt de aanroepende code een object [Promise]
    // dankzij [await] krijgt u het [text]-antwoord van de server in plaats van zijn promise
    // const text = await response.text();
    // console.log("text response=", text);
    // return text;
  } catch (error: any) {
    // we zijn hier omdat de server een foutcode heeft gestuurd [404 Not Found, ...] met een lege body - we tonen de fout om de structuur ervan te zien
    // of omdat de client [fetch] een exception heeft geworpen (netwerk onbereikbaar, ...)
    // we tonen de structuur van de fout
    console.log(sprintf("fetch error in json=%j, %s", error, time(fetchStart)));
    console.log("fetch error in javascript=", typeof (error), error);
    // we werpen het ontvangen foutbericht
    throw error.message;
  }
}

// de hoofdfunctie voert de asynchrone functie [initSession] uit
async function main(): Promise<void> {
  try {
    console.log("HTTP request to the server in progress ---------------------------------------------");
    const response = await initSession();
    console.log("success ---------------------------------------------");
    console.log("response=", response, typeof (response))
  } catch (error) {
    console.log("error ---------------------------------------------");
    console.log("error=", error, typeof (error));
  }
}

// test
main();

// hulpfunctie die tijd en duur toont
function time(start?: moment.Moment): string {
  // tijd van dit moment
  const now = moment(Date.now());
  // tijdopmaak
  let result = "time=" + now.format("HH:mm:ss:SSS");
  // moet een duur worden berekend?
  if (start) {
    const duration = now.valueOf() - start.valueOf();
    const milliseconds = duration % 1000;
    const seconds = Math.floor(duration / 1000);
    // opmaak van tijd + duur
    result = result + sprintf(", duration= %s second(s) and %s milliseconds", seconds, milliseconds);
  }
  // result
  return result;
}

    'use strict';
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';

// standaardconfiguratie van axios
axios.defaults.timeout = 2000;
axios.defaults.baseURL = 'http://localhost:3000/';

// sessie initialiseren
async function initSession(axios: AxiosInstance): Promise<any> {
  // opties van het HTTP-verzoek [get /main.php?action=init-session&type=json]
  const options: AxiosRequestConfig = {
    // [TypeScript-update] axios.request() accepteert, wat de typering betreft, slechts één
    // configuratieargument: de URL moet dus in [options.url] staan
    // in plaats van als aparte 1e parameter van request()
    url: 'main.php',
    method: "GET",
    // URL-parameters
    params: {
      action: 'init-session',
      type: 'x'
    }
  };
  // uitvoering van het HTTP-verzoek [get /main.php?action=init-session&type=json]
  try {
    // asynchroon verzoek
    const response = await axios.request(options);
    // response is het hele HTTP-antwoord van de server (HTTP-headers + het antwoord zelf)
    // we tonen dit antwoord om de structuur ervan te zien
    // console.log("axios response=", response);
    // het antwoord van de server staat in [response.data]
    return response.data;
  } catch (error: any) {
    // we zijn hier omdat de server een foutcode heeft gestuurd [404 Not Found, 500 Internal Server Error, ...]
    // de parameter [error] is een uitzonderingsinstantie - die kan verschillende vormen aannemen
    // we tonen die om de structuur ervan te zien
    // console.log("axios error=", typeof (error), error);
    if (error.response) {
      // de server heeft een fout gemeld in de HTTP-status, maar heeft ook een antwoord gestuurd
      // dat dan in [error.response.data] staat
      // we weten dat de server JSON-antwoorden met de structuur {action, status, response} stuurt
      // en dat bij een fout de foutmelding in [response] staat
      return error.response.data;
    } else {
      // we werpen de fout
      throw error;
    }
  }
}

// de hoofdfunctie voert de asynchrone functie [initSession] uit
async function main(): Promise<void> {
  try {
    console.log("HTTP request to the server in progress ---------------------------------------------");
    const response = await initSession(axios);
    console.log("success ---------------------------------------------");
    console.log("response=", response, typeof (response))
  } catch (error: any) {
    console.log("error ---------------------------------------------");
    console.log("error=", error.message);
  }
}

    'use strict';
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';
import qs from 'qs'

// configuratie van axios
axios.defaults.timeout = 2000;
axios.defaults.baseURL = 'http://localhost:3000/';


// sessie initialiseren
async function initSession(axios: AxiosInstance): Promise<any> {
  // opties van het HTTP-verzoek [get /main.php?action=init-session&type=json]
  const options: AxiosRequestConfig = {
    // [TypeScript-update] URL opgenomen in [options.url] (de getypeerde axios.request() accepteert slechts één argument)
    url: 'main.php',
    method: "GET",
    // URL-parameters
    params: {
      action: 'init-session',
      type: 'json'
    }
  };
  try {
    // uitvoering van het HTTP-verzoek [get /main.php?action=init-session&type=json]
    const response = await axios.request(options);
    // het antwoord van de server staat in [response.data]
    return response.data;
  } catch (error: any) {
    // antwoord van de server
    if (error.response) {
      // het JSON-antwoord staat in [error.response.data]
      return error.response.data;
    } else {
      // we werpen de fout opnieuw
      throw error;
    }
  }
}

async function authenticateUser(axios: AxiosInstance, user: string, password: string): Promise<any> {
  // opties van het HTTP-verzoek [post /main.php?action=authenticate-user]
  const options: AxiosRequestConfig = {
    // [TypeScript-update] URL opgenomen in [options.url]
    url: 'main.php',
    method: "POST",
    headers: {
      'Content-type': 'application/x-www-form-urlencoded',
    },
    // body van de POST
    data: qs.stringify({
      user: user,
      password: password
    }),
    // URL-parameters
    params: {
      action: 'authenticate-user'
    }
  };
  try {
    // uitvoering van het HTTP-verzoek [post /main.php?action=authenticate-user]
    const response = await axios.request(options);
    // het antwoord van de server staat in [response.data]
    return response.data;
  } catch (error: any) {
    // antwoord van de server
    if (error.response) {
      // het JSON-antwoord staat in [error.response.data]
      return error.response.data;
    } else {
      // we werpen de fout opnieuw
      throw error;
    }
  }
}

// de hoofdfunctie voert de asynchrone functies één voor één uit
async function main(): Promise<void> {
  try {
    // init-session
    console.log("init-session action in progress ---------------------------------------------");
    const response1 = await initSession(axios);
    console.log("success ---------------------------------------------");
    console.log("response=", response1);
    // authenticate-user
    console.log("authenticate-user action in progress ---------------------------------------------");
    const response2 = await authenticateUser(axios, 'admin', 'admin');
    console.log("success ---------------------------------------------");
    console.log("response=", response2)
  } catch (error) {
    console.log("error ---------------------------------------------");
    console.log("error=", error);
  }
}

    • Regels 10-14: de actie [authenticate-user] is mislukt. De reden is als volgt:
      • De actie [init-session] heeft de sessiecookie niet naar de server gestuurd. Dit is de identificatie van de client bij de server. Omdat de client deze identificatie niet heeft verstuurd, is hij onbekend bij de server. In zijn antwoord heeft de server een sessiecookie meegestuurd. Dit zegt tegen de client: „Hier is uw identificatie voor uw toekomstige verzoeken”;
      • maar de actie [authenticate-user] heeft de sessiecookie evenmin naar de server gestuurd. Dat komt doordat [init-session] hem niet heeft opgeslagen. De server herkent de client dus niet en weet niet dat die eerder een JSON-sessie heeft gestart, en geeft dat aan de client door;

    De oplossing voor dit probleem is het beheren van de sessiecookie die de server verstuurt:

    • Het allereerste verzoek aan de server moet de door de server verzonden sessiecookie opslaan;
    • volgende verzoeken moeten deze sessiecookie ophalen en naar de server sturen om zich te authenticeren;

    'use strict';
import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

// configuratie van axios
axios.defaults.timeout = 2000;
axios.defaults.baseURL = 'http://localhost:3000/';

// sessiecookie
const sessionCookieName: string = "connect.sid";
let sessionCookie: string = '';

// sessie initialiseren
async function initSession(axios: AxiosInstance): Promise<any> {
  // opties van het HTTP-verzoek [get /main.php?action=init-session&type=json]
  const options: AxiosRequestConfig = {
    // [TypeScript-update] URL opgenomen in [options.url]
    url: 'main.php',
    method: "GET",
    // URL-parameters
    params: {
      action: 'init-session',
      type: 'json'
    }
  };
  // uitvoering van het HTTP-verzoek
  return await getRemoteData(axios, options);
}

async function authenticateUser(axios: AxiosInstance, user: string, password: string): Promise<any> {
  // opties van het HTTP-verzoek [post /main.php?action=authenticate-user]
  const options: AxiosRequestConfig = {
    // [TypeScript-update] URL opgenomen in [options.url]
    url: 'main.php',
    method: "POST",
    headers: {
      'Content-type': 'application/x-www-form-urlencoded',
    },
    // body van de POST
    data: qs.stringify({
      user: user,
      password: password
    }),
    // URL-parameters
    params: {
      action: 'authenticate-user'
    }
  };
  // uitvoering van het HTTP-verzoek
  return await getRemoteData(axios, options);
}

async function getRemoteData(axios: AxiosInstance, options: AxiosRequestConfig): Promise<any> {
  // voor de sessiecookie
  if (!options.headers) {
    options.headers = {} as any;
  }
  (options.headers as any).Cookie = sessionCookie;
  // uitvoering van het HTTP-verzoek
  let response: AxiosResponse;
  try {
    // asynchroon verzoek
    response = await axios.request(options);
  } catch (error: any) {
    // de parameter [error] is een uitzonderingsinstantie - die kan verschillende vormen aannemen
    if (error.response) {
      // het antwoord van de server staat in [error.response]
      response = error.response;
    } else {
      // we werpen de fout opnieuw
      throw error;
    }
  }
  // response is the whole HTTP response of the server (HTTP headers + the response itself)
  // we halen de sessiecookie op als die bestaat
  const setCookie = response.headers['set-cookie'];
  if (setCookie) {
    // setCookie is een array
    // we zoeken de sessiecookie in deze array
    let found = false;
    let i = 0;
    while (!found && i < setCookie.length) {
      // we zoeken de sessiecookie
      const results = RegExp('^(' + sessionCookieName + '.+?);').exec(setCookie[i]);
      if (results) {
        // we slaan de sessiecookie op
        // eslint-disable-next-line require-atomic-updates
        sessionCookie = results[1];
        // gevonden
        found = true;
      } else {
        // volgend element
        i++;
      }
    }
  }
  // het antwoord van de server staat in [response.data]
  return response.data;
}


// de hoofdfunctie voert de asynchrone functies één voor één uit
async function main(): Promise<void> {
  try {
    // init-session
    console.log("init-session action in progress ----------------");
    const response1 = await initSession(axios);
    console.log("success --------------------------");
    console.log("response=", response1);
    // authenticate-user
    console.log("authenticate-user action in progress ---------");
    const response2 = await authenticateUser(axios, 'admin', 'admin');
    console.log("success -------------------");
    console.log("response=", response2)
  } catch (error: any) {
    console.log("error ------------------");
    console.log("error=", error.message);
  }
}

// test
main();

    Dit script brengt initSession en authenticateUser onder in een gemeenschappelijke functie, [getRemoteData], die de sessiecookie expliciet afhandelt (hier connect.sid, zie regel 10); dit is essentieel omdat de server authenticatie en simulaties aan een HTTP-sessie koppelt:

    • regel 10: de naam van de sessiecookie waarnaar wordt gezocht. Deze hangt af van de gebruikte webserver. De NestJS-server die in deze cursus wordt gebruikt, gebruikt een sessiecookie met de naam [connect.sid];
    • regel 53: de functie [getRemoteData] verzorgt het verzenden van verzoeken en het ontvangen van het antwoord van de server. Ze beheert ook de sessiecookie. Ze heeft twee parameters:
      • het object [AxiosInstance], dat de algemene werking van HTTP-verzoeken met Axios afhandelt. Dit object en andere worden verkregen via de import op regel 2;
      • het object [AxiosRequestConfig], dat de opties voor een specifiek verzoek definieert;
    • Regels 54-58: [options.headers] stelt de header in van het HTTP-verzoek dat naar de server wordt gestuurd. Als er een sessiecookie bestaat, moet die in deze header worden opgenomen. Regel 58 doet dit. Na afloop van het eerste verzoek slaat [getRemoteData] deze cookie op in de globale variabele die op regel 11 is gedefinieerd. Tijdens het eerste verzoek is de waarde een lege tekenreeks. Bij volgende verzoeken is dit de tekenreeks van de sessiecookie die de server als antwoord op het eerste verzoek heeft gestuurd.
    • Regel 63: het HTTP-verzoek wordt uitgevoerd;
    • regel 76: in de antwoordheader wordt gecontroleerd of er een regel „set-cookie” is. Deze regel is alleen aanwezig als het vorige verzoek van de client geen regel „Cookie” in zijn headers bevatte. Wanneer de server een HTTP-verzoek van de client ontvangt dat de regel „Cookie” bevat, stuurt hij de sessiecookie niet meer mee, omdat hij weet dat de client die al heeft. Het is aan de client om die bij elk nieuw verzoek opnieuw te verzenden. Dat gebeurt op regel 58;
    • regel 77: als de server de header [Set-Cookie] heeft verzonden, moet de inhoud ervan worden opgehaald. Deze header kan meerdere cookies bevatten in de vorm [cookie_name=value]. Regels 77-95 worden slechts één keer uitgevoerd tijdens de uitvoering van het script. Zoals gezegd stuurt de server de sessiecookie niet mee als antwoord op verzoeken die met diezelfde sessiecookie zijn geauthenticeerd;
    • regel 84: we zoeken expliciet naar de cookie waarvan de naam op regel 10 is opgeslagen;
    • regel 88: als die wordt gevonden, wordt hij opgeslagen in de globale variabele die op regel 11 is gedefinieerd;

    Dit mechanisme zorgt ervoor dat:

    • de sessiecookie wordt opgehaald uit het antwoord op het eerste HTTP-verzoek;
    • die vervolgens systematisch bij elk nieuw verzoek naar de server wordt gestuurd, zodat de server zijn client kan herkennen;

// basis-URL van de server voor de belastingberekening (zie http/README.md)

// draait: zonder die server mislukken de verzoeken met "fetch failed", wat normaal is
const baseUrl: string = 'http://localhost:3000/main.php';

// voorbeeld 1: verzoek met een maximale wachttijd dankzij AbortSignal.timeout()
async function requestWithMaximumDelay(): Promise<void> {
  console.log("--- request with AbortSignal.timeout ---");
  try {
    const response = await fetch(baseUrl + '?action=init-session&type=json', {
      // na 2 seconden wordt het verzoek automatisch geannuleerd
      signal: AbortSignal.timeout(2000)
    });
    const data = await response.json();
    console.log("response =", data);
  } catch (error: any) {
    // als de wachttijd wordt overschreden,
    // is [error.name] "TimeoutError" of "AbortError", afhankelijk van het geval
    console.log("error.name =", error.name, ", message =", error.message);
  }
}

// voorbeeld 2: handmatige annulering van een verzoek, bijvoorbeeld omdat
// de gebruiker zich bedacht heeft (bijv.: hij typt opnieuw in een zoekveld)
async function manuallyCancellableRequest(): Promise<void> {
  console.log("--- manually cancelled request ---");
  const controller = new AbortController();

  // we plannen een "functionele" annulering na 500 ms
  // (in een echte applicatie zou dit bijvoorbeeld een klik op een knop "annuleren" zijn)
  setTimeout(() => {
    console.log("manual cancellation triggered");
    controller.abort();
  }, 10);

  try {
    const response = await fetch(baseUrl + '?action=init-session&type=json', {
      signal: controller.signal
    });
    const data = await response.json();
    console.log("response =", data);
  } catch (error: any) {
    console.log("error.name =", error.name, ", message =", error.message);
  }
}

async function main(): Promise<void> {
  await requestWithMaximumDelay();
  await manuallyCancellableRequest();
}
